BuildingUI MonoBehaviour

Gère l'affichage UI ancré au-dessus d'un objet 3D dans le monde (World To Screen Space) et réarrange ses modules enfants.

using System.Collections.Generic;
using UnityEngine;

namespace GAME
{
    [RequireComponent(typeof(RectTransform))]
    public class BuildingUI : GameObjectUI, UILanguage
    {
        public GameObject ModulesContainer;
        private List<UIBuildingModule> moduleUIs = new List<UIBuildingModule>();

        [Tooltip("L'objet 3D dans le monde au-dessus duquel l'UI doit s'afficher")]
        public GameObject UIPivotPoint;

        public override void Start()
        {
            base.Start();
        }

        public virtual void Update()
        {
            UpdateUIPosition();
        }

        public void AddModule(UIBuildingModule module)
        {
            if (module == null) return;
            module.transform.SetParent(ModulesContainer.transform, false);
            moduleUIs.Add(module);
            RearrangeModules();
        }

        public void RemoveModule(UIBuildingModule module)
        {
            if (module == null) return;
            moduleUIs.Remove(module);
            if (module.gameObject != null) Destroy(module.gameObject);
            RearrangeModules();
        }

        private void RearrangeModules()
        {
            for (int i = 0; i < moduleUIs.Count; i++)
                moduleUIs[i].transform.SetSiblingIndex(i);

            this.gameObject.RebuidLayout();
            UpdateUIPosition();
        }

        private void UpdateUIPosition()
        {
            if (UIPivotPoint == null) return;

            Canvas rootCanvas = GetComponentInParent<Canvas>().rootCanvas;
            Camera uiCamera = rootCanvas.worldCamera;

            Vector2 screenPosition = RectTransformUtility.WorldToScreenPoint(
                uiCamera != null ? uiCamera : null,
                UIPivotPoint.transform.position);

            if (rootCanvas.renderMode == RenderMode.ScreenSpaceOverlay)
            {
                transform.position = screenPosition;
            }
            else if (uiCamera != null)
            {
                RectTransformUtility.ScreenPointToLocalPointInRectangle(
                    transform.parent as RectTransform,
                    screenPosition,
                    uiCamera,
                    out Vector2 localPosition);

                transform.localPosition = localPosition;
            }
            else if (rootCanvas.renderMode == RenderMode.WorldSpace)
            {
                transform.position = UIPivotPoint.transform.position;
            }
        }

        public void Language() { }
    }
}

UIWindowBuildingModule UILanguage

Base pour les modules affichés dans une fenêtre avec gestion du texte localisé et cycle de validation (Apply/Cancel).

using TMPro;
using UnityEngine;

namespace GAME
{
    public class UIWindowBuildingModule : MonoBehaviour, UILanguage
    {
        public string Title;
        public TextMeshProUGUI TitleField;

        public virtual void Initialize(BuildingModule module)
        {
            Langs.AddHandler(Language);
        }

        public virtual void Language()
        {
            TitleField.text = Langs.GetText(Title);
        }

        public virtual bool NeedValidation() => false;

        public virtual void Apply() { }

        public virtual void Cancel() { }

        public void OnDestroy()
        {
            Langs.RemoveHandler(Language);
        }
    }
}

UIWindowBuildingModulesContainer UIWindowStack

Conteneur de fenêtre gérant une liste de modules, recalculant le layout dynamique et gérant la validation globale.

using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;

namespace GAME
{
    public class UIWindowBuildingModulesContainer : UIWindowStack
    {
        public Building Building;
        public Text BuildingText;
        public GameObject ContainerModules;
        private List<UIWindowBuildingModule> modules = new List<UIWindowBuildingModule>();

        public void InitializeContainer(Building building)
        {
            Building = building;

            bool needValidation = false;
            foreach (var module in modules)
            {
                if (module.NeedValidation())
                {
                    needValidation = true;
                    break;
                }
            }

            if (YesButton != null) YesButton.gameObject.SetActive(needValidation);
            if (NoButton != null) NoButton.gameObject.SetActive(needValidation);

            Langs.AddHandler(Language);
        }

        public override int Open(IDictionary<string, object> p)
        {
            if (ContainerModules != null)
            {
                RectTransform containerRect = ContainerModules.GetComponent<RectTransform>();

                foreach (Transform child in ContainerModules.transform)
                {
                    LayoutRebuilder.ForceRebuildLayoutImmediate(child.GetComponent<RectTransform>());
                }

                LayoutRebuilder.ForceRebuildLayoutImmediate(containerRect);

                RectTransform panel = this.GetComponent<RectTransform>().GetWorkingRect();
                LayoutRebuilder.ForceRebuildLayoutImmediate(panel);
            }

            return base.Open(p);
        }

        protected override void CloseWindow(UIWINDOWSTACK_EVENT evt)
        {
            if (evt == UIWINDOWSTACK_EVENT.YES)
            {
                foreach (var module in modules) module.Apply();
            }
            else if (evt == UIWINDOWSTACK_EVENT.NO)
            {
                foreach (var module in modules) module.Cancel();
            }

            base.CloseWindow(evt);
        }

        public void AddModule(UIWindowBuildingModule module)
        {
            if (module == null) return;
            modules.Add(module);
        }

        public void RemoveModule(UIWindowBuildingModule module)
        {
            if (module == null) return;
            modules.Remove(module);
        }

        public void ClearModules()
        {
            for (int i = modules.Count - 1; i >= 0; i--)
            {
                if (modules[i] != null && modules[i].gameObject != null)
                    Destroy(modules[i].gameObject);
            }
            modules.Clear();
        }

        public void Language()
        {
            BuildingText.text = Langs.GetText(Building.Type.ToString());
        }

        public void OnDestroy()
        {
            Langs.RemoveHandler(Language);
        }
    }
}

InfoWindowBuilding UILanguage

Fenêtre d'information sur un bâtiment permettant d'activer/désactiver le bâtiment et de gérer la zone d'effet (cercle persistant).

using TMPro;
using UnityEngine;
using UnityEngine.UI;

namespace GAME
{
    public class InfoWindowBuilding : UIWindowStack, UILanguage
    {
        public TextMeshProUGUI Title;
        public Building Building;
        public bool UseEnableButton;
        public GameObject Enable;
        public GameObject Disable;
        public bool BuildingEnable;
        public Toggle TogglePersistentCircle;

        public void InitializeEnable()
        {
            if (UseEnableButton)
            {
                if (Enable != null)
                {
                    Enable.GetComponent<Button>().onClick.AddListener(EnableBulding);
                    Disable.GetComponent<Button>().onClick.AddListener(DisableBulding);
                }
            }
            else
            {
                if (Enable != null) Enable.SetActive(false);
                if (Disable != null) Disable.SetActive(false);
            }

            if (TogglePersistentCircle != null)
                TogglePersistentCircle.onValueChanged.AddListener(
                    delegate { PersistentCircleToggleChange(); });
        }

        public virtual void Initialize(Building building)
        {
            Building = building;

            if (UseEnableButton)
            {
                BuildingEnable = Building.Enable;
                if (Enable != null)
                {
                    Enable.SetActive(BuildingEnable);
                    Disable.SetActive(!BuildingEnable);
                }
            }
            else
                BuildingEnable = Building.Enable;

            if (TogglePersistentCircle != null)
                TogglePersistentCircle.isOn = building.PersistentCircle;

            InitializeEnable();
            Langs.AddHandler(Language);
        }

        public virtual void Work() { }

        public virtual void EnableBulding()
        {
            BuildingEnable = false;
            Enable.SetActive(false);
            Disable.SetActive(true);
        }

        public virtual void DisableBulding()
        {
            BuildingEnable = true;
            Enable.SetActive(true);
            Disable.SetActive(false);
        }

        public virtual void OnEnable() { }
        public virtual void OnDisable() { }

        public void PersistentCircleToggleChange()
        {
            Building.SetPersistentCircle(TogglePersistentCircle.isOn);
        }

        public virtual void Language()
        {
            if (Building != null)
                Title.text = Langs.GetText(Building.Name);
        }

        public void OnDestroy()
        {
            Langs.RemoveHandler(Language);
        }
    }
}
